1. tasks table update migration

Schema::table('tasks', function (Blueprint $table) {
    $table->date('due_date')->nullable()->after('priority');
    $table->time('due_time')->nullable()->after('due_date');

    $table->boolean('is_recurring')->default(false)->after('due_time');
    $table->enum('recurring_type', ['daily', 'weekly', 'monthly'])->nullable()->after('is_recurring');
    $table->date('recurring_end_date')->nullable()->after('recurring_type');

    $table->unsignedInteger('position')->default(0)->after('recurring_end_date');
});

5. Drag & Drop reorder method

TaskController.php

public function reorder(Request $request)
{
    $request->validate([
        'tasks' => 'required|array',
        'tasks.*.id' => 'required|exists:tasks,id',
        'tasks.*.position' => 'required|integer',
    ]);

    foreach ($request->tasks as $taskItem) {
        Task::where('id', $taskItem['id'])
            ->where('user_id', auth()->id())
            ->update([
                'position' => $taskItem['position'],
            ]);
    }

    return response()->json([
        'success' => true,
        'message' => 'Task order updated.',
    ]);
}

    if (sortable) {
        new Sortable(sortable, {
            animation: 180,
            ghostClass: 'opacity-50',
            onEnd: function () {

                document.querySelectorAll('#taskSortable .task-card').forEach((item, index) => {
                    tasks.push({

                        position: index + 1

                    method: "PATCH",
                    headers: {
                        "Content-Type": "application/json",
                        "X-CSRF-TOKEN": "{{ csrf_token() }}"
                    },
                    body: JSON.stringify({
                        tasks: tasks
                    })
                });
            }
        });
    }
</script>
10. Recurring task auto-create logic

Simple version: task completed হলে next recurring task create হবে।

TaskController@updateStatus এ status completed হলে add করো:

if ($request->status === 'completed' && $task->is_recurring) {
    $nextDueDate = match ($task->recurring_type) {
        'daily' => $task->due_date?->copy()->addDay(),
        'weekly' => $task->due_date?->copy()->addWeek(),
        'monthly' => $task->due_date?->copy()->addMonth(),
        default => null,
    };

    if ($nextDueDate && (!$task->recurring_end_date || $nextDueDate->lte($task->recurring_end_date))) {
        $newTask = $task->replicate();
        $newTask->status = 'pending';
        $newTask->due_date = $nextDueDate;
        $newTask->position = Task::where('user_id', auth()->id())->max('position') + 1;
        $newTask->created_at = now();
        $newTask->updated_at = now();
        $newTask->save();

        $newTask->labels()->sync($task->labels->pluck('id')->toArray());

        TaskHistory::create([
            'task_id' => $newTask->id,
            'user_id' => auth()->id(),
            'action' => 'Recurring Task Created',
            'changes' => 'Created from completed recurring task: ' . $task->title,
        ]);
    }
}
11. History update for recurring/deadline/order

Task update এ due date/recurring change track করতে পারো:

$changes = [];

if ($task->wasChanged('due_date')) {
    $changes['due_date'] = [
        'old' => $task->getOriginal('due_date'),
        'new' => $task->due_date,
    ];
}

if ($task->wasChanged('is_recurring') || $task->wasChanged('recurring_type')) {
    $changes['recurring'] = [
        'old' => $task->getOriginal('recurring_type') ?? 'Not recurring',
        'new' => $task->recurring_type ?? 'Not recurring',
    ];
}

if (!empty($changes)) {
    TaskHistory::create([
        'task_id' => $task->id,
        'user_id' => auth()->id(),
        'action' => 'Task Schedule Updated',
        'changes' => json_encode($changes),
    ]);
}
